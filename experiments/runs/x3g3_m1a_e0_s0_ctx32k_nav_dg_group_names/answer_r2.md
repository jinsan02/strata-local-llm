class InvalidGroupName(ValueError):
    """
    An invalid dependency group name was provided.
    Group names must follow the same rules as the core metadata `Name` field.

    .. versionadded:: 26.2
    """

    def __init__(self, name: str) -> None:
        super().__init__(f"Invalid dependency group name: {name!r}")
